cmake_minimum_required(VERSION 3.0.2)
project(fastlio_planner_bridge)
find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS})
add_executable(fastlio_planner_odometry src/fastlio_planner_odometry.cpp)
target_compile_features(fastlio_planner_odometry PRIVATE cxx_std_14)
target_link_libraries(fastlio_planner_odometry ${catkin_LIBRARIES})
install(TARGETS fastlio_planner_odometry RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
