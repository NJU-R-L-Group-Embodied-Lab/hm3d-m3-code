cmake_minimum_required(VERSION 3.22)
project(ekf LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(ekf
  src/ekf_node_vio_timesync_with_acc_pub.cpp
  src/conversion.cpp)
target_include_directories(ekf PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(ekf rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs)
target_link_libraries(ekf Eigen3::Eigen)
add_executable(check_initialization src/check_initialization.cpp src/conversion.cpp)
target_include_directories(check_initialization PRIVATE include)
ament_target_dependencies(check_initialization rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs)
target_link_libraries(check_initialization Eigen3::Eigen)
enable_testing()
foreach(scenario prior_settling settled translation stale_pose invalid_pose moving_start imu_gap)
  add_test(NAME initialization_${scenario} COMMAND check_initialization ${scenario})
endforeach()
install(TARGETS ekf RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME}
  FILES_MATCHING PATTERN "*.launch.py" PATTERN "*.launch.xml")
ament_package()
